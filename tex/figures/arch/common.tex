% \archcomp[box style]{name}{x,y}{icon}{title}{subtitle}
\newcommand{\archcomp}[6][archbox]{%
  \node[#1] (#2) at (#3) {};
  \pic at ($(#2.west)+(0.42,0)$) {#4};
  \node[archtext] at ($(#2.west)+(0.82,0)$) {\textbf{#5}\\#6};
}

}

  \node[font=\sffamily\tiny, text=black!75, anchor=north, align=center, inner sep=1pt] (#1-l) at ($(#1.south)+(0,-0.02)$) {#3};
}
